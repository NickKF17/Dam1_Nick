#!/bin/bash


ARCHIVO="datos.txt"


if [ $# -eq 0 ]; then
    echo "Error: Por favor, introduce al menos un parámetro."
    exit 1
fi

for param in "$@"; do
    echo "$param" >> "$ARCHIVO"
done

echo "--- Parámetros guardados en $ARCHIVO ---"
echo "--- Contenido del fichero en orden inverso: ---"
tac "$ARCHIVO"
